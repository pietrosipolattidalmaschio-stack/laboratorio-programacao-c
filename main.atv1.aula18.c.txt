#include <stdio.h>

int exibirMenu() {
    int opcao;
    printf("\n===== CALCULADORA =====\n");
    printf("1- Soma\n");
    printf("2- Subtração\n");
    printf("3- Divisão\n");
    printf("4- Multiplicação\n");
    printf("0- Sair\n");
    printf("Opção escolhida: ");
    scanf("%d", &opcao);
    return opcao;
}
int soma(int a, int b) {
    return a + b;
}
int subtracao(int a, int b) {
    return a - b;
}
int divisao(int a, int b) {
    return a / b;
}
int multiplicacao(int a, int b) {
    return a * b;
}

int main(void) {
    
    int opcao, a, b, resultado;
    
    do {
        
        opcao = exibirMenu();
        
        if (opcao == 0) {
            printf("Programa encerrado.\n");
            break;
        }
        
        if (opcao < 1 || opcao > 4) {
            printf("Erro. Digite uma opção válida");
            continue;
        }
        
        printf("Digite o primeiro número: ");
        scanf("%d", &a);
        printf("Digite o segundo número: ");
        scanf("%d", &b);
        
        switch (opcao) {
            case 1:
                resultado = soma(a, b);
                printf("Resultado da soma: %d\n", resultado);
                break;
            
            case 2:
                resultado = subtracao(a, b);
                printf("Resultado da subtração: %d\n", resultado);
                break;
            
            case 3:
                if (b == 0) {
                    printf("Divisão por zero inválida. Digite outro número.");
                    scanf("%d", &b);
                }
                resultado = divisao(a, b);
                printf("Resultado da divisão: %d\n", resultado);
                break;
            
            case 4:
                resultado = multiplicacao(a, b);
                printf("Resultado da multiplicação: %d\n", resultado);
                break;
        }
    } while (opcao != 0);
    
    return 0;
}
