#include <stdio.h>

int calcularFatorial(int n) {
    int resultado = 1;
    
    for (int i = 1; i <= n; i++) {
        resultado = resultado * i;
    }
    return resultado;
}

int main() {
    printf("===== CALCULADORA DE FATORIAL =====\n");
    int n, resultado;
    
    printf("Digite um número: ");
    scanf("%d", &n);
    if (n < 0 || n > 12) {
        printf("Número inválido. Digite um número entre 0 e 12: ");
        scanf("%d", &n);
    }
    
    printf("\nFatorial de %d é: %d", n, calcularFatorial(n));
    return 0;
}
